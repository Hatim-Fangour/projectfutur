// 1. USER MANAGEMENT

enum UserRole {
  OWNER
  MANAGER
  STAFF
  RECEPTIONIST
  THERAPIST
}

model Customer {
  id           String   @id @default(cuid())
  fullName     String
  email        String   @unique
  phone        String?
  company      String?
  
  // Location
  country      String?
  state        String?
  city         String?
  address      String?
  
  // Profile
  pictureURL   String?
  
  // Timestamps
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt
  
  // Relations (matching your TypeScript type)
  notes        CustomerNote[]        // notes?: CustomerNoteType[]
  appointments Appointment[]         // appointments?: CustomerAppointmentType[]
  services     CustomerService[]     // services?: PricingPlan[]
  progress     ProgressSession[]     // progress?: CustomerProgressSessionType[]
  
  @@index([email])
  @@index([fullName])
  @@index([phone])
  @@map("customers")
}

// ===================================================
// 3. SERVICES & TREATMENTS
// ===================================================

enum ServiceCategory {
  MASSAGE
  FACIAL
  BODY_TREATMENT
  MANICURE_PEDICURE
  HAIR_REMOVAL
  WELLNESS
  PACKAGE
}

model Service {
  id           String          @id @default(cuid())
  name         String
  description  String?         @db.Text
  category     ServiceCategory
  
  // Pricing
  price        Decimal         @db.Decimal(10, 2)
  discountPrice Decimal?       @db.Decimal(10, 2)
  
  // Duration in minutes
  duration     Int
  bufferTime   Int             @default(0)
  
  // Settings
  isActive     Boolean         @default(true)
  color        String?
  imageURL     String?
  
  // Requirements
  requiresDeposit Boolean      @default(false)
  depositAmount   Decimal?     @db.Decimal(10, 2)
  
  // Timestamps
  createdAt    DateTime        @default(now())
  updatedAt    DateTime        @updatedAt
  
  // Relations
  appointments AppointmentService[]
  packages     PackageService[]
  
  @@index([category])
  @@index([isActive])
  @@map("services")
}

// Service packages (e.g., "Spa Day Package")
model Package {
  id           String          @id @default(cuid())
  name         String
  description  String?         @db.Text
  price        Decimal         @db.Decimal(10, 2)
  
  // Settings
  isActive     Boolean         @default(true)
  validityDays Int?            // How long package is valid
  
  // Timestamps
  createdAt    DateTime        @default(now())
  updatedAt    DateTime        @updatedAt
  
  // Relations
  services     PackageService[]
  
  @@map("packages")
}

model PackageService {
  id         String   @id @default(cuid())
  packageId  String
  serviceId  String
  quantity   Int      @default(1)
  
  package    Package  @relation(fields: [packageId], references: [id], onDelete: Cascade)
  service    Service  @relation(fields: [serviceId], references: [id], onDelete: Cascade)
  
  @@map("package_services")
}


// ===================================================
// 4. APPOINTMENTS & BOOKINGS
// ===================================================

enum AppointmentStatus {
  SCHEDULED
  CONFIRMED
  IN_PROGRESS
  COMPLETED
  CANCELLED
  NO_SHOW
}

enum AppointmentType {
  SERVICE
  CLASS
  EVENT
  REMINDER
}

model Appointment {
  id           String            @id @default(cuid())
  type         AppointmentType   @default(SERVICE)
  
  // Customer & Staff
  customerId   String
  staffId      String?
  
  // Scheduling
  startTime    DateTime
  endTime      DateTime
  bufferEnd    DateTime?         // End time including buffer
  allDay       Boolean           @default(false)
  
  // Details
  status       AppointmentStatus @default(SCHEDULED)
  room         String?
  notes        String?           @db.Text
  color        String?
  
  // Reminders
  reminderSent Boolean           @default(false)
  reminderText String?
  
  // Event-specific
  eventName    String?
  location     String?
  videoLink    String?
  
  // Timestamps
  createdAt    DateTime          @default(now())
  updatedAt    DateTime          @updatedAt
  cancelledAt  DateTime?
  
  // Relations
  customer     Customer          @relation(fields: [customerId], references: [id], onDelete: Cascade)
  staff        User?             @relation(fields: [staffId], references: [id])
  services     AppointmentService[]
  
  @@index([customerId])
  @@index([staffId])
  @@index([startTime])
  @@index([status])
  @@map("appointments")
}

// Many-to-many: Appointments can have multiple services
model AppointmentService {
  id            String      @id @default(cuid())
  appointmentId String
  serviceId     String
  
  // Pricing at time of booking (in case prices change)
  price         Decimal     @db.Decimal(10, 2)
  
  appointment   Appointment @relation(fields: [appointmentId], references: [id], onDelete: Cascade)
  service       Service     @relation(fields: [serviceId], references: [id])
  
  @@map("appointment_services")
}


// ===================================================
// 5. CLASSES & GROUP SESSIONS
// ===================================================

enum ClassStatus {
  SCHEDULED
  ONGOING
  COMPLETED
  CANCELLED
}

model Class {
  id           String       @id @default(cuid())
  title        String
  description  String?      @db.Text
  
  // Scheduling
  startTime    DateTime
  endTime      DateTime
  recurrence   String?      // JSON: {type: 'weekly', days: [1,3,5]}
  
  // Capacity
  maxSeats     Int
  enrolledCount Int         @default(0)
  
  // Pricing
  cost         Decimal      @db.Decimal(10, 2)
  
  // Details
  status       ClassStatus  @default(SCHEDULED)
  instructor   String?
  room         String?
  imageURL     String?
  color        String?
  
  // Timestamps
  createdAt    DateTime     @default(now())
  updatedAt    DateTime     @updatedAt
  
  // Relations
  enrollments  ClassEnrollment[]
  
  @@index([startTime])
  @@map("classes")
}

model ClassEnrollment {
  id         String   @id @default(cuid())
  classId    String
  customerId String
  
  // Payment
  paid       Boolean  @default(false)
  paidAmount Decimal? @db.Decimal(10, 2)
  
  // Status
  attended   Boolean  @default(false)
  
  // Timestamps
  enrolledAt DateTime @default(now())
  
  class      Class    @relation(fields: [classId], references: [id], onDelete: Cascade)
  customer   Customer @relation(fields: [customerId], references: [id], onDelete: Cascade)
  
  @@unique([classId, customerId])
  @@map("class_enrollments")
}


// ===================================================
// 6. INVOICES & PAYMENTS
// ===================================================

enum InvoiceStatus {
  DRAFT
  SENT
  PAID
  OVERDUE
  CANCELLED
}

enum PaymentMethod {
  CASH
  CARD
  BANK_TRANSFER
  MOBILE_PAYMENT
  GIFT_CARD
}

model Invoice {
  id           String        @id @default(cuid())
  invoiceNumber String       @unique
  
  // Customer
  customerId   String
  
  // Amounts
  subtotal     Decimal       @db.Decimal(10, 2)
  tax          Decimal       @db.Decimal(10, 2) @default(0)
  discount     Decimal       @db.Decimal(10, 2) @default(0)
  total        Decimal       @db.Decimal(10, 2)
  
  // Status
  status       InvoiceStatus @default(DRAFT)
  
  // Dates
  issueDate    DateTime      @default(now())
  dueDate      DateTime?
  paidDate     DateTime?
  
  // Notes
  notes        String?       @db.Text
  
  // Timestamps
  createdAt    DateTime      @default(now())
  updatedAt    DateTime      @updatedAt
  
  // Relations
  customer     Customer      @relation(fields: [customerId], references: [id])
  items        InvoiceItem[]
  payments     Payment[]
  
  @@index([customerId])
  @@index([status])
  @@index([invoiceNumber])
  @@map("invoices")
}

model InvoiceItem {
  id          String  @id @default(cuid())
  invoiceId   String
  
  description String
  quantity    Int     @default(1)
  unitPrice   Decimal @db.Decimal(10, 2)
  total       Decimal @db.Decimal(10, 2)
  
  invoice     Invoice @relation(fields: [invoiceId], references: [id], onDelete: Cascade)
  
  @@map("invoice_items")
}

model Payment {
  id            String        @id @default(cuid())
  invoiceId     String
  
  amount        Decimal       @db.Decimal(10, 2)
  method        PaymentMethod
  
  // Details
  transactionId String?
  notes         String?
  
  // Timestamps
  paidAt        DateTime      @default(now())
  
  invoice       Invoice       @relation(fields: [invoiceId], references: [id])
  
  @@index([invoiceId])
  @@map("payments")
}


// ===================================================
// 7. INVENTORY MANAGEMENT
// ===================================================

enum ProductCategory {
  RETAIL
  SUPPLIES
  EQUIPMENT
  CONSUMABLES
}

model Product {
  id           String          @id @default(cuid())
  name         String
  description  String?         @db.Text
  category     ProductCategory
  
  // SKU & Barcode
  sku          String?         @unique
  barcode      String?
  
  // Pricing
  costPrice    Decimal?        @db.Decimal(10, 2)
  sellPrice    Decimal?        @db.Decimal(10, 2)
  
  // Stock
  stockQuantity Int            @default(0)
  minStockLevel Int?
  unit         String?         // e.g., "piece", "bottle", "kg"
  
  // Details
  brand        String?
  supplier     String?
  imageURL     String?
  
  // Status
  isActive     Boolean         @default(true)
  
  // Timestamps
  createdAt    DateTime        @default(now())
  updatedAt    DateTime        @updatedAt
  
  // Relations
  logs         InventoryLog[]
  
  @@index([sku])
  @@index([category])
  @@map("products")
}

enum InventoryAction {
  PURCHASE
  SALE
  ADJUSTMENT
  DAMAGE
  RETURN
}

model InventoryLog {
  id          String          @id @default(cuid())
  productId   String
  userId      String
  
  action      InventoryAction
  quantity    Int
  
  // Before/After
  previousQty Int
  newQty      Int
  
  // Details
  notes       String?         @db.Text
  reference   String?         // Invoice number, PO number, etc.
  
  // Timestamps
  createdAt   DateTime        @default(now())
  
  product     Product         @relation(fields: [productId], references: [id])
  user        Customer            @relation(fields: [userId], references: [id])
  
  @@index([productId])
  @@index([createdAt])
  @@map("inventory_logs")
}


// ===================================================
// 8. MEMBERSHIPS & LOYALTY
// ===================================================

enum MembershipStatus {
  ACTIVE
  EXPIRED
  CANCELLED
  SUSPENDED
}

model MembershipPlan {
  id           String   @id @default(cuid())
  name         String
  description  String?  @db.Text
  
  // Pricing
  price        Decimal  @db.Decimal(10, 2)
  
  // Duration
  durationDays Int      // e.g., 30, 90, 365
  
  // Benefits
  benefits     String?  @db.Text // JSON string
  
  // Settings
  isActive     Boolean  @default(true)
  
  // Timestamps
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt
  
  // Relations
  memberships  Membership[]
  
  @@map("membership_plans")
}

model Membership {
  id          String           @id @default(cuid())
  customerId  String
  planId      String
  
  // Dates
  startDate   DateTime
  endDate     DateTime
  
  // Status
  status      MembershipStatus @default(ACTIVE)
  
  // Auto-renewal
  autoRenew   Boolean          @default(false)
  
  // Timestamps
  createdAt   DateTime         @default(now())
  updatedAt   DateTime         @updatedAt
  cancelledAt DateTime?
  
  customer    Customer         @relation(fields: [customerId], references: [id])
  plan        MembershipPlan   @relation(fields: [planId], references: [id])
  
  @@index([customerId])
  @@index([status])
  @@map("memberships")
}


// ===================================================
// 9. REVIEWS & FEEDBACK
// ===================================================

model Review {
  id         String   @id @default(cuid())
  customerId String
  
  rating     Int      // 1-5
  comment    String?  @db.Text
  
  // Optional: Link to specific appointment/service
  appointmentId String?
  
  // Status
  isPublished Boolean @default(true)
  
  // Timestamps
  createdAt  DateTime @default(now())
  updatedAt  DateTime @updatedAt
  
  customer   Customer @relation(fields: [customerId], references: [id])
  
  @@index([customerId])
  @@index([rating])
  @@map("reviews")
}
